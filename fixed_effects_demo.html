fixed_effects_demo.R
1#c====================================================================
2# Panel Data Regression - Fixed Effects Model (FEM) Simulation
3# =====================================================================
4# Step 0: Install and load required packages
5if (!require("plm")) install.packages("plm")
6library(plm)
7
8# ------------------------------------------------------------------------------------------------------------
9# 1. SIMULATE PANEL DATA
10# ------------------------------------------------------------------------------------------------------------
11set.seed(310) # For reproducibility
12N <- 4 # 4 entities (e.g., Airlines: Delta, United, American, US Airways)
13T <- 15 # 15 time periods (e.g., Years: 1990 to 2004)
14total_obs <- N * T
15
16# Create Entity (i) and Time (t) identifiers
17firm_id <- factor(rep(1:N, each = T))
18year <- rep(1990:(1990 + T - 1), times = N)
19
20# Define True Entity Fixed Effects (unobserved heterogeneity alpha_i)
21# Delta = 0 (base), United = -120, American = -80, US Airways = 40
22true_alpha <- c(0, -120, -80, 40)
23alpha_i <- rep(true_alpha, each = T)
24
25# Generate Explanatory Variable X (Output: Ticket Miles)
26# Notice that Output is correlated with alpha_i (Well-managed firms produce more output)
27X_output <- round(runif(total_obs, min = 200, max = 1000) + 0.5 * alpha_i)
28# Generate Idiosyncratic Noise (epsilon_it)
29epsilon <- rnorm(total_obs, mean = 0, sd = 15)
30
31# Generate Dependent Variable Y (Total Cost)
32# True Model: Cost = 50 + 0.10 * Output + alpha_i + epsilon_it
33true_beta2 <- 0.10
34Y_cost <- 50 + true_beta2 * X_output + alpha_i + epsilon
35
36# Assemble into a clean data frame
37panel_data <- data.frame(
38 firm_id = firm_id,
39 year = year,
40 Cost = Y_cost,
41 Output = X_output
42)
43
44head(panel_data, 10) # Preview data
45
46
47# ------------------------------------------------------------------------------------------------------------
48# 2. ESTIMATION METHOD 1: POOLED OLS (Flawed Benchmark)
49# ------------------------------------------------------------------------------------------------------------
50pooled_ols <- lm(Cost ~ Output, data = panel_data)
51cat("\n--- POOLED OLS RESULTS ---\n")
52summary(pooled_ols)
53
54# ------------------------------------------------------------------------------------------------------------
55# 3. ESTIMATION METHOD 2: LSDV (Least Squares Dummy Variable)
56# ------------------------------------------------------------------------------------------------------------
57lsdv_model <- lm(Cost ~ Output + firm_id, data = panel_data)
58cat("\n--- LSDV FIXED EFFECTS RESULTS ---\n")
59summary(lsdv_model)
60
61# ------------------------------------------------------------------------------------------------------------
62# 4. ESTIMATION METHOD 3: WITHIN-GROUP ESTIMATOR (Time-Demeaning using plm)
63# ------------------------------------------------------------------------------------------------------------
64p_data <- pdata.frame(panel_data, index = c("firm_id", "year"))
65fe_model <- plm(Cost ~ Output, data = p_data, model = "within")
66cat("\n--- WITHIN-GROUP FIXED EFFECTS (plm) RESULTS ---\n")
67summary(fe_model)
68
69cat("\nIs LSDV slope identical to Within-Group slope?\n")
70cat("LSDV Slope:", coef(lsdv_model)["Output"], "\n")
71cat("WG Slope: ", coef(fe_model)["Output"], "\n")
72
73# ------------------------------------------------------------------------------------------------------------
74# 5. RECOVERING INDIVIDUAL ENTITY INTERCEPTS
75# ------------------------------------------------------------------------------------------------------------
76cat("\n--- RECOVERED FIXED EFFECTS INTERCEPTS (beta_1i) ---\n")
77fixed_effects_intercepts <- fixef(fe_model)
78print(fixed_effects_intercepts)
79
80# ------------------------------------------------------------------------------------------------------------
81# 6. MANUAL TIME-DEMEANING (Hand Calculation Verification)
82# ------------------------------------------------------------------------------------------------------------
83panel_data$mean_Cost <- ave(panel_data$Cost, panel_data$firm_id, FUN = mean)
84panel_data$mean_Output <- ave(panel_data$Output, panel_data$firm_id, FUN = mean)
85panel_data$demeaned_Cost <- panel_data$Cost - panel_data$mean_Cost
86panel_data$demeaned_Output <- panel_data$Output - panel_data$mean_Output
87
88cat("\nSum of demeaned Cost for Firm 1:", sum(panel_data$demeaned_Cost[panel_data$firm_id == 1]), "\n")
89
90manual_wg <- lm(demeaned_Cost ~ demeaned_Output - 1, data = panel_data)
91cat("\n--- MANUAL TIME-DEMEANED OLS SLOPE ---\n")
92coef(manual_wg)
93
94# ------------------------------------------------------------------------------------------------------------
95# 7. DIAGNOSTIC HYPOTHESIS TESTING
96# ------------------------------------------------------------------------------------------------------------
97cat("\n--- REDUNDANT FIXED EFFECTS TEST (pFtest) ---\n")
98pFtest(fe_model, pooled_ols)
99
100re_model <- plm(Cost ~ Output, data = p_data, model = "random")
101
102cat("\n--- HAUSMAN SPECIFICATION TEST (phtest) ---\n")
103phtest(fe_model, re_model)
